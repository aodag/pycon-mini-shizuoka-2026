#+begin_src python
  def bankaccount(probe: Generator[None, int|str, None], balance: int=0) -> Generator[int, Message, None]:
      try:
          while True:
              probe.send(balance)
              msg = yield
              match msg:
                  case Deposit(amount):
                      balance += amount
                  case Withdraw(amount):
                      balance -= amount
      except GeneratorExit:
          probe.send("shutdown bankaccount")
          return balance
#+end_src
